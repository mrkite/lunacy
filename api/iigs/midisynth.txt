WaveList struct {
	topKey: u8;
	configuration: u8;
	channel: u8;
	detune: u8;
	waveAddressA: u8;
	waveSizeA: u8;
	volumeA: u8;

	waveSizeB: u8;
	volumeB: u8;
	octaveB: u8;
	semitoneB: u8;
	fineTuneB: u8;

MIDIEnvelope struct {
	attackLevel: u8;
	attackRate: u8;
	decay1Level: u8;
	decay1Rate: u8;
	decay2Level: u8;
	decay2Rate: u8;
	sustainLevel: u8;
	decay3Rate: u8;
	release1Level: u8;
	release1Rate: u8;
	release2Level: u8;
	release2Rate: u8;
	releast3Rate: u8;
	decayGain: u8;
	velocityGain: u8;
	pitchBend: u8;
}

MIDIInstrument struct {
	gen1Envelope: MIDIEnvelope;
	gen1Waves: WaveList[8];
	gen2Envelope: MIDIEnvelope;
	gen2Waves: WaveList[8];
}
